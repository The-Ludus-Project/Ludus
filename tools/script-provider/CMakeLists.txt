# Tests use only the installed surface and project-owned cooked inputs.
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
set(cooked "${PROJECT_SOURCE_DIR}/out/script-provider/cooked")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/script-provider" verify
    RESULT_VARIABLE cooked_result OUTPUT_QUIET)
if(NOT cooked_result EQUAL 0)
    message(FATAL_ERROR "Prepare paired tools using scripts/script-provider bootstrap")
endif()
file(READ "${cooked}/current.json" pointer)
string(JSON key GET "${pointer}" key)
set(cooked "${cooked}/${key}")
add_executable(ludus_behavior_acceptance consumer/main.cpp)
target_include_directories(ludus_behavior_acceptance PRIVATE "${cooked}")
target_link_libraries(ludus_behavior_acceptance PRIVATE Ludus::Behavior)
ludus_apply_project_defaults(ludus_behavior_acceptance)
enable_testing()
add_test(NAME ludus_behavior_cook COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_cook.py")
set_tests_properties(ludus_behavior_cook PROPERTIES TIMEOUT 120 LABELS "scripting;behavior;tooling")
if(EMSCRIPTEN)
    target_link_options(ludus_behavior_acceptance PRIVATE -sEXIT_RUNTIME=1 -sENVIRONMENT=web,node
        -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1 -sSTACK_SIZE=1048576 -sSTACK_OVERFLOW_CHECK=2)
endif()
add_test(NAME ludus_behavior_acceptance COMMAND ludus_behavior_acceptance)
set_tests_properties(ludus_behavior_acceptance PROPERTIES TIMEOUT 60 LABELS "scripting;behavior")
file(READ "${PROJECT_SOURCE_DIR}/out/script-provider/safety/current.json" safety_pointer)
string(JSON safety_key GET "${safety_pointer}" key)
# Compile the override once; each executable links its own copy of the object and allocation-failure state.
add_library(ludus_behavior_fault_allocator OBJECT fault_allocator.cpp)
target_link_libraries(ludus_behavior_fault_allocator PRIVATE Ludus::FoundationBase)
ludus_apply_project_defaults(ludus_behavior_fault_allocator)
add_executable(ludus_behavior_safety safety.cpp)
target_include_directories(ludus_behavior_safety PRIVATE "${PROJECT_SOURCE_DIR}/out/script-provider/safety/${safety_key}")
target_link_libraries(ludus_behavior_safety PRIVATE ludus_behavior_fault_allocator Ludus::Behavior Ludus::FoundationLogging)
ludus_apply_project_defaults(ludus_behavior_safety)
if(EMSCRIPTEN)
    target_link_options(ludus_behavior_safety PRIVATE -sEXIT_RUNTIME=1 -sENVIRONMENT=web,node
        -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1 -sSTACK_SIZE=1048576 -sSTACK_OVERFLOW_CHECK=2)
    configure_file(shell.html "${CMAKE_CURRENT_BINARY_DIR}/index.html" COPYONLY)
endif()
set(safety_runner)
if(EMSCRIPTEN)
    set(safety_runner --runner ${CMAKE_CROSSCOMPILING_EMULATOR})
endif()
add_test(NAME ludus_behavior_safety COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_safety.py"
    --build "${PROJECT_BINARY_DIR}" --exe "$<TARGET_FILE:ludus_behavior_safety>"
    --cook "${PROJECT_SOURCE_DIR}/out/script-provider/safety/${safety_key}" ${safety_runner})
set_tests_properties(ludus_behavior_safety PROPERTIES TIMEOUT 120 LABELS "scripting;behavior;safety")
add_test(NAME ludus_behavior_profiles COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_profiles.py")
set_tests_properties(ludus_behavior_profiles PROPERTIES TIMEOUT 60 LABELS "scripting;behavior;tooling")
add_test(NAME ludus_behavior_evidence COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_evidence.py")
set_tests_properties(ludus_behavior_evidence PROPERTIES TIMEOUT 30 LABELS "scripting;behavior;tooling")
# Exercise the exact private allocator contract separately from SDK-only safety.
add_executable(ludus_behavior_allocator_contract allocator_contract.cpp)
target_include_directories(ludus_behavior_allocator_contract PRIVATE "${PROJECT_SOURCE_DIR}/modules/runtime/scripting/src")
target_link_libraries(ludus_behavior_allocator_contract PRIVATE ludus_behavior_fault_allocator ludus_runtime_scripting Ludus::FoundationLogging)
ludus_apply_project_defaults(ludus_behavior_allocator_contract)
if(EMSCRIPTEN)
    target_link_options(ludus_behavior_allocator_contract PRIVATE -sENVIRONMENT=web,node -sEXIT_RUNTIME=1
        -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1)
endif()
add_test(NAME ludus_behavior_allocator_contract COMMAND ludus_behavior_allocator_contract)
set_tests_properties(ludus_behavior_allocator_contract PROPERTIES TIMEOUT 30 LABELS "scripting;behavior;safety")
if(LUDUS_BUILD_TESTS)
    find_package(Catch2 3 REQUIRED CONFIG)
    add_executable(ludus_behavior_tests tests.cpp)
    target_include_directories(ludus_behavior_tests PRIVATE "${cooked}")
    target_compile_definitions(ludus_behavior_tests PRIVATE "LUDUS_BEHAVIOR_PACKAGE=\"${cooked}/behavior.lupack\"")
    target_link_libraries(ludus_behavior_tests PRIVATE Ludus::Behavior Ludus::FoundationContainers Catch2::Catch2WithMain)
    ludus_apply_project_defaults(ludus_behavior_tests)
    ludus_enable_test_exceptions(ludus_behavior_tests)
    add_test(NAME ludus_behavior_tests COMMAND ludus_behavior_tests)
    set_tests_properties(ludus_behavior_tests PROPERTIES TIMEOUT 60 LABELS "unit;scripting;behavior")
endif()
