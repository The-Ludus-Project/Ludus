cmake_minimum_required(VERSION 3.29)
project(LightingRoom LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Ludus CONFIG REQUIRED COMPONENTS GraphicsRenderer Platform FoundationTime)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_executable(lighting_room MACOSX_BUNDLE src/game.cpp)
target_link_libraries(lighting_room PRIVATE Ludus::GraphicsRenderer Ludus::Platform Ludus::FoundationTime)
ludus_apply_app_policy(lighting_room)
target_compile_options(lighting_room PRIVATE -Wall -Wextra -Werror)
set(world "${CMAKE_CURRENT_SOURCE_DIR}/authoring/lighting.world.json")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/lighting_world.h"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/cook_world.py"
        "${world}" "${CMAKE_CURRENT_BINARY_DIR}/lighting_world.h"
    DEPENDS "${world}" cook_world.py VERBATIM)
target_sources(lighting_room PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/lighting_world.h")
target_include_directories(lighting_room PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set(forward_source "${Ludus_DIR}/render_graph/templates/forward_lit")
set(forward_staged "${CMAKE_CURRENT_BINARY_DIR}/forward_lit")
foreach(file main.lurender.json rasterize.lurender.json main_shaders/geometry.slang main_shaders/color.slang)
    configure_file("${forward_source}/${file}" "${forward_staged}/${file}" COPYONLY)
endforeach()
set(LUDUS_DEFAULT_RENDER_GRAPH_PROFILES "[\"metal\"]")
configure_file("${forward_source}/main.lushader.json.in" "${forward_staged}/main.lushader.json" @ONLY)
ludus_compile_renderer_shaders(TARGET lighting_room SHADERS renderer_flat renderer_composite)
ludus_compile_render_graph(TARGET lighting_room NAME forward_lit MANIFEST "${forward_staged}/main.lushader.json")
if(BUILD_TESTING)
    add_test(NAME lighting_room_qualification COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/qualify.py" $<TARGET_FILE:lighting_room>)
    set_tests_properties(lighting_room_qualification PROPERTIES TIMEOUT 120)
endif()

include(cmake/GameRelease.cmake)
