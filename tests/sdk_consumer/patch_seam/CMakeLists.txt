cmake_minimum_required(VERSION 3.29)
project(LudusPatchSeamConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Ludus CONFIG REQUIRED COMPONENTS GeometryContent FoundationGeometry GraphicsRenderer FoundationThreading FoundationTime)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_executable(ludus_sdk_patch_seam capture.cpp)
target_link_libraries(ludus_sdk_patch_seam PRIVATE Ludus::FoundationThreading Ludus::FoundationTime)
target_compile_options(ludus_sdk_patch_seam PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
ludus_compile_renderer_shaders(TARGET ludus_sdk_patch_seam SHADERS renderer_composite)
include(graph.cmake)
ludus_compile_patch_seam_graph(ludus_sdk_patch_seam "${Ludus_DIR}/render_graph/templates/forward_lit")
foreach(mode geometry raster)
    add_test(NAME ludus_sdk_patch_seam_${mode} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify.py"
        "${mode}" "${CMAKE_CURRENT_BINARY_DIR}/${mode}" "$<TARGET_FILE:ludus_sdk_patch_seam>")
    set_tests_properties(ludus_sdk_patch_seam_${mode} PROPERTIES TIMEOUT 240)
endforeach()
