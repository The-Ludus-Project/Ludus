cmake_minimum_required(VERSION 3.29)
project(LudusLargeWorldSky LANGUAGES CXX)
find_package(Ludus CONFIG REQUIRED COMPONENTS GraphicsRenderer Resources FoundationTime FoundationThreading)
add_executable(skylight MACOSX_BUNDLE capture.cpp)
target_compile_features(skylight PRIVATE cxx_std_23)
target_compile_options(skylight PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-exceptions)
set_target_properties(skylight PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(skylight PRIVATE Ludus::GraphicsRenderer Ludus::Resources Ludus::FoundationTime Ludus::FoundationThreading)
ludus_compile_renderer_shaders(TARGET skylight SHADERS renderer_composite)

# Stage the installed default graph with its output set to scene-linear for probe comparisons.
get_filename_component(sdk "${Ludus_DIR}/../../.." ABSOLUTE)
set(staged "${CMAKE_CURRENT_BINARY_DIR}/linear")
file(COPY "${sdk}/share/Ludus/render-graphs/default/" DESTINATION "${staged}")
file(READ "${staged}/main.lurender.json" graph)
string(JSON count LENGTH "${graph}" configurations)
math(EXPR last "${count} - 1")
foreach(configuration RANGE ${last})
    string(JSON count LENGTH "${graph}" configurations ${configuration} nodes)
    math(EXPR last_node "${count} - 1")
    foreach(node RANGE ${last_node})
        string(JSON kind GET "${graph}" configurations ${configuration} nodes ${node} kind)
        if(kind STREQUAL "ToneMap")
            string(JSON graph SET "${graph}" configurations ${configuration} nodes ${node} enabled false)
        endif()
    endforeach()
endforeach()
file(WRITE "${staged}/main.lurender.json" "${graph}\n")
ludus_compile_render_graph(TARGET skylight NAME skylight MANIFEST "${staged}/main.lushader.json" RUNTIME_PACKAGE)
set_source_files_properties("${skylight_RUNTIME_PACKAGE}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
target_sources(skylight PRIVATE "${skylight_RUNTIME_PACKAGE}")
foreach(control open exposed)
    foreach(file scene.luscn camera.txt)
        set(input "${LUDUS_SKY_CASES}/${control}/${file}")
        set_source_files_properties("${input}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${control}")
        target_sources(skylight PRIVATE "${input}")
    endforeach()
endforeach()
